#[cfg(target_os = "android")]
fn set_pty_size(fd: std::os::fd::RawFd, cols: u16, rows: u16) -> std::io::Result<()> {
    let size = libc::winsize {
        ws_row: rows,
        ws_col: cols,
        ws_xpixel: 0,
        ws_ypixel: 0,
    };
    let rc = unsafe { libc::ioctl(fd, libc::TIOCSWINSZ as _, &size) };
    if rc < 0 {
        return Err(std::io::Error::last_os_error());
    }
    Ok(())
}

#[cfg(target_os = "android")]
fn open_pty(
    cols: u16,
    rows: u16,
) -> Result<(tokio::fs::File, StdFile, Stdio, Stdio, Stdio)> {
    let master_fd = unsafe {
        libc::posix_openpt(libc::O_RDWR | libc::O_NOCTTY | libc::O_CLOEXEC)
    };
    if master_fd < 0 {
        return Err(std::io::Error::last_os_error()).context("posix_openpt");
    }
    if unsafe { libc::grantpt(master_fd) } < 0 {
        let error = std::io::Error::last_os_error();
        unsafe { libc::close(master_fd); }
        return Err(error).context("grantpt");
    }
    if unsafe { libc::unlockpt(master_fd) } < 0 {
        let error = std::io::Error::last_os_error();
        unsafe { libc::close(master_fd); }
        return Err(error).context("unlockpt");
    }
    let mut name = [0 as libc::c_char; 128];
    let pts_rc = unsafe { libc::ptsname_r(master_fd, name.as_mut_ptr(), name.len()) };
    if pts_rc != 0 {
        unsafe { libc::close(master_fd); }
        return Err(std::io::Error::from_raw_os_error(pts_rc)).context("ptsname_r");
    }
    let slave_fd = unsafe {
        libc::open(
            name.as_ptr(),
            libc::O_RDWR | libc::O_NOCTTY | libc::O_CLOEXEC,
        )
    };
    if slave_fd < 0 {
        let error = std::io::Error::last_os_error();
        unsafe { libc::close(master_fd); }
        return Err(error).context("open PTY slave");
    }
    if let Err(error) = set_pty_size(slave_fd, cols, rows) {
        unsafe {
            libc::close(slave_fd);
            libc::close(master_fd);
        }
        return Err(error).context("set initial PTY size");
    }
    let resize_fd = unsafe { libc::dup(master_fd) };
    if resize_fd < 0 {
        let error = std::io::Error::last_os_error();
        unsafe {
            libc::close(slave_fd);
            libc::close(master_fd);
        }
        return Err(error).context("dup PTY master");
    }
    let master = unsafe { StdFile::from_raw_fd(master_fd) };
    let resize = unsafe { StdFile::from_raw_fd(resize_fd) };
    let slave = unsafe { StdFile::from_raw_fd(slave_fd) };
    let slave_out = slave.try_clone().context("clone PTY stdout")?;
    let slave_err = slave.try_clone().context("clone PTY stderr")?;
    Ok((
        tokio::fs::File::from_std(master),
        resize,
        Stdio::from(slave),
        Stdio::from(slave_out),
        Stdio::from(slave_err),
    ))
}

async fn handle_pty(stream: &mut TcpStream, stream_id: u32, payload: &[u8]) -> Result<()> {
    let request: PtyOpenRequest = decode(payload)?;
    #[cfg(not(target_os = "android"))]
    {
        let _ = (stream, stream_id, request);
        bail!("PTY is only supported by the Android daemon");
    }
    #[cfg(target_os = "android")]
    {
        let (master, resize, stdin, stdout, stderr) =
            open_pty(request.cols.max(1), request.rows.max(1))?;
        let mut command = Command::new(&request.program);
        command
            .args(&request.args)
            .env("TERM", &request.term)
            .env("COLORTERM", "truecolor")
            .stdin(stdin)
            .stdout(stdout)
            .stderr(stderr);
        unsafe {
            command.pre_exec(|| {
                if libc::setsid() < 0 {
                    return Err(std::io::Error::last_os_error());
                }
                if libc::ioctl(libc::STDIN_FILENO, libc::TIOCSCTTY as _, 0) < 0 {
                    return Err(std::io::Error::last_os_error());
                }
                Ok(())
            });
        }
        apply_identity(&mut command, request.identity)?;
        let mut child = command
            .spawn()
            .with_context(|| format!("spawn PTY program {}", request.program))?;
        let pid = child.id().unwrap_or(0);
        write_frame(
            stream,
            &frame(FrameKind::PtyOpenAck, stream_id, &PtyOpenAck { pid })?,
        )
        .await?;
        let (mut master_read, mut master_write) = tokio::io::split(master);
        let mut buffer = vec![0u8; 64 * 1024];
        loop {
            tokio::select! {
                read = master_read.read(&mut buffer) => {
                    match read {
                        Ok(0) => break,
                        Ok(n) => {
                            write_frame(
                                stream,
                                &raw_frame(FrameKind::PtyOutput, stream_id, buffer[..n].to_vec()),
                            ).await?;
                        }
                        Err(error) if error.raw_os_error() == Some(libc::EIO) => break,
                        Err(error) => return Err(error).context("read PTY master"),
                    }
                }
                incoming = read_frame(stream) => {
                    let incoming = incoming?;
                    if incoming.stream_id != stream_id {
                        bail!("unexpected stream {} while PTY {stream_id} is active", incoming.stream_id);
                    }
                    match incoming.kind {
                        FrameKind::PtyInput => master_write.write_all(&incoming.payload).await?,
                        FrameKind::PtyResize => {
                            let size: PtyResize = decode(&incoming.payload)?;
                            set_pty_size(resize.as_raw_fd(), size.cols.max(1), size.rows.max(1))?;
                        }
                        FrameKind::PtyClose => {
                            let _ = child.kill().await;
                            break;
                        }
                        kind => bail!("unexpected PTY frame {kind:?}"),
                    }
                }
            }
        }
        let status = child.wait().await?;
        let exit = PtyExit { code: status.code() };
        write_frame(stream, &frame(FrameKind::PtyExit, stream_id, &exit)?).await?;
        Ok(())
    }
}
